    <script>
        document.getElementById('adminLoginForm').addEventListener('submit', function(event) {
            event.preventDefault(); // Prevent form from submitting normally

            const username = document.getElementById('username').value;
            const password = document.getElementById('password').value;
            const errorMessage = document.getElementById('errorMessage');

            // --- IMPORTANT SECURITY WARNING ---
            // This is a "client-side" check and is NOT secure.
            // Anyone can view the page source to find the password.
            // For a real application, this check MUST be done on a server.
            
            // For demonstration, we'll use a hardcoded username and password.
            const correctUsername = "admin";
            const correctPassword = "lifewood_admin_2024!"; // Example password

            if (username === correctUsername && password === correctPassword) {
                // If credentials are correct, redirect to the dashboard.
                errorMessage.textContent = '';
                window.location.href = 'dashboard.html';
            } else {
                // If credentials are wrong, show an error message.
                errorMessage.textContent = 'Invalid username or password.';
            }
        });
    </script>
